{% load csp %} {% comment %} Shared search + sort toolbar for contact lists (Network's Contacts section, Archived Contacts). Included with `target` = the id of the container that holds the `[data-searchable]` items. Zero server round trip: everything a row needs to match or sort by is baked into its own data-* attributes at render time, so the JS never re-parses rendered, title-cased text. Groups are optional. An item's group is `.closest("[data-group]")`, falling back to the target container itself — Network's warmth `
` mark themselves `data-group` (per-group counts + collapse-while-searching), while Archived has no groups and the script quietly treats the whole list as one. {% endcomment %}

{% script %} (function () { // The toolbar renders ABOVE the items it searches (better UX than // burying it under 130 cards), which means at the moment this inline // script executes, the rest of #{{ target }} hasn't been parsed into // the DOM yet — querySelectorAll("[data-searchable]") would find zero. // Deferring the actual wiring past DOMContentLoaded fixes it without // moving the markup. if (document.readyState === "loading") { document.addEventListener("DOMContentLoaded", init); } else { init(); } function init() { var root = document.getElementById("{{ target }}"); if (!root) return; var input = root.querySelector("[data-csearch-q]"); var sortSel = root.querySelector("[data-csearch-sort]"); var globalEmpty = root.querySelector("[data-csearch-empty]"); var globalEmptyQ = root.querySelector("[data-csearch-empty-q]"); var items = Array.prototype.slice.call(root.querySelectorAll("[data-searchable]")); if (!input || !sortSel || !items.length) return; var openBeforeSearch = null; // null = not currently searching function groupOf(el) { return el.closest("[data-group]") || root; } function numOr(v, fallback) { return v === "" || v == null ? fallback : +v; } function applySearch() { var q = input.value.trim().toLowerCase(); // EVERY WORD, IN ANY ORDER — not one contiguous run of characters. // `indexOf` meant the query had to appear exactly as typed, so a // haystack reading "Garrett Roach Bank of America" answered "garrett // roach" and refused "roach garrett", refused "garrett bofa", and // refused "garrett roach" on the double space. Nobody types a CRM // search expecting to reproduce a word order they cannot see.
